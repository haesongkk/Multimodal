(event EventTick (DeltaSeconds)
  (bind _returnvalue (HUD|GetOwningPlayerController))
  (bind _returnvalue_1 (HUD|GetOwningPawn))
  (CallFunction|SyncJournal)
  (Utilities|IsValid _returnvalue_1
    (:"Is Valid"
      (if (not (Variables|디폴트|GetComplete))
        (Variables|디폴트|SetNearby 0)
        (Variables|디폴트|SetNearestDistance 175.0)
        (Variables|디폴트|SetPrompt "코코가 남긴 흔적을 찾아보자.")
        (if (< (Transformation|GetHorizontalDistanceTo _returnvalue_1 (Variables|디폴트|GetPotActor)) (Variables|디폴트|GetNearestDistance))
          (Variables|디폴트|SetNearestDistance (Transformation|GetHorizontalDistanceTo _returnvalue_1 (Variables|디폴트|GetPotActor)))
          (Variables|디폴트|SetNearby 1)
          (Variables|디폴트|SetPrompt (select (Variables|디폴트|GetPotSeen) "[E] 넘어진 화분 다시 확인" "[E] 넘어진 화분 조사")))
        (if (< (Transformation|GetHorizontalDistanceTo _returnvalue_1 (Variables|디폴트|GetFurActor)) (Variables|디폴트|GetNearestDistance))
          (Variables|디폴트|SetNearestDistance (Transformation|GetHorizontalDistanceTo _returnvalue_1 (Variables|디폴트|GetFurActor)))
          (Variables|디폴트|SetNearby 2)
          (Variables|디폴트|SetPrompt (select (Variables|디폴트|GetFurSeen) "[E] 고양이 털 다시 확인" "[E] 고양이 털 조사")))
        (if (< (Transformation|GetHorizontalDistanceTo _returnvalue_1 (Variables|디폴트|GetWindowActor)) (Variables|디폴트|GetNearestDistance))
          (Variables|디폴트|SetNearestDistance (Transformation|GetHorizontalDistanceTo _returnvalue_1 (Variables|디폴트|GetWindowActor)))
          (Variables|디폴트|SetNearby 3)
          (Variables|디폴트|SetPrompt (select (Variables|디폴트|GetWindowSeen) "[E] 열린 창문 다시 확인" "[E] 열린 창문 조사")))
        (if (< (Transformation|GetHorizontalDistanceTo _returnvalue_1 (Variables|디폴트|GetDoorActor)) (Variables|디폴트|GetNearestDistance))
          (Variables|디폴트|SetNearestDistance (Transformation|GetHorizontalDistanceTo _returnvalue_1 (Variables|디폴트|GetDoorActor)))
          (Variables|디폴트|SetNearby 4)
          (Variables|디폴트|SetPrompt "[E] 현관문 확인"))
        (if (< (Transformation|GetHorizontalDistanceTo _returnvalue_1 (Variables|디폴트|GetSnackActor)) (Variables|디폴트|GetNearestDistance))
          (Variables|디폴트|SetNearestDistance (Transformation|GetHorizontalDistanceTo _returnvalue_1 (Variables|디폴트|GetSnackActor)))
          (Variables|디폴트|SetNearby 5)
          (Variables|디폴트|SetPrompt (select (Variables|디폴트|GetSnackSeen) "[E] 먹다 남은 간식 다시 확인" "[E] 먹다 남은 간식 조사")))
        (if (< (Transformation|GetHorizontalDistanceTo _returnvalue_1 (Variables|디폴트|GetPawprintsActor)) (Variables|디폴트|GetNearestDistance))
          (Variables|디폴트|SetNearestDistance (Transformation|GetHorizontalDistanceTo _returnvalue_1 (Variables|디폴트|GetPawprintsActor)))
          (Variables|디폴트|SetNearby 6)
          (Variables|디폴트|SetPrompt (select (Variables|디폴트|GetPawprintsSeen) "[E] 작은 발자국 다시 확인" "[E] 작은 발자국 조사")))
        (if (Variables|디폴트|GetPopup)
          (if (Game|Player|WasInputKeyJustPressed _returnvalue "SpaceBar")
            (CallFunction|Confirm))
          (else
            (Pawn|Input|AddMovementInput _returnvalue_1 "0,1,0" (select (Game|Player|IsInputKeyDown _returnvalue "W") 1.0 0.0))
            (Pawn|Input|AddMovementInput _returnvalue_1 "0,-1,0" (select (Game|Player|IsInputKeyDown _returnvalue "S") 1.0 0.0))
            (Pawn|Input|AddMovementInput _returnvalue_1 "1,0,0" (select (Game|Player|IsInputKeyDown _returnvalue "A") 1.0 0.0))
            (Pawn|Input|AddMovementInput _returnvalue_1 "-1,0,0" (select (Game|Player|IsInputKeyDown _returnvalue "D") 1.0 0.0))
            (if (Game|Player|WasInputKeyJustPressed _returnvalue "E")
              (CallFunction|Interact))))
        (if (Variables|디폴트|GetDoorOpen)
          (Variables|디폴트|SetDoorAngle (select (> (+ (Variables|디폴트|GetDoorAngle) (* DeltaSeconds 100.0)) 90.0) 90.0 (+ (Variables|디폴트|GetDoorAngle) (* DeltaSeconds 100.0))))
          (Transformation|SetActorRotation (Variables|디폴트|GetDoorActor) (Math|Rotator|MakeRotator 0.0 0.0 (- (Variables|디폴트|GetDoorAngle)))))
        (if (and (>= (Variables|디폴트|GetDoorAngle) 90.0) (< (Transformation|GetHorizontalDistanceTo _returnvalue_1 (Variables|디폴트|GetExitActor)) 65.0))
          (Variables|디폴트|SetComplete true)
          (Variables|디폴트|SetTitle "집 조사 완료")
          (Variables|디폴트|SetMessage "다섯 가지 흔적을 모두 수첩에 담았어. 코코를 찾아 골목으로 가 보자!")
          (Variables|디폴트|SetPopup true)
          (CallFunction|RefreshPopup))))))
